```mermaid
@startuml
autonumber

actor Test as T
participant "HomomorphicEmbedder" as HE
participant "keygen()" as KG
participant "find_qnr()" as FQ
participant "ToyHE" as TH
participant "bit_to_element()" as B2E
participant "element_to_bit()" as E2B

T -> HE: HomomorphicEmbedder(bits=256)
HE -> KG: keygen(bits)
KG --> HE: p, q, n
HE -> TH: ToyHE(n)
TH --> HE: he
HE -> FQ: find_qnr(n,p,q)
FQ --> HE: z
HE --> T: embedder

T -> T: 生成 pixels
T -> T: message="Hi" -> msg_bits

T -> HE: embed(pixels, msg_bits)

loop for each pixel
    HE -> HE: A = pixel mod n
    alt A==0 or gcd(A,n)!=1
        HE -> HE: 跳过该像素
    else 可嵌入
        HE -> TH: enc(A)
        TH --> HE: ct_a, r_a

        HE -> B2E: bit_to_element(bit,n,z)
        B2E --> HE: B

        HE -> TH: enc(B)
        TH --> HE: ct_b, r_b

        HE -> TH: homo_multiply((ct_a,r_a),(ct_b,r_b))
        TH --> HE: ct_ab, r_ab

        HE -> HE: 保存 packet={ct,r,original_pixel}
    end
end

HE --> T: packets

T -> HE: extract(packets)

loop for each packet
    HE -> TH: dec(ct_ab,r_ab)
    TH --> HE: ab=A*B mod n

    HE -> HE: A = original_pixel
    HE -> HE: A_inv = A^{-1} mod n
    HE -> HE: B = ab * A_inv mod n

    HE -> E2B: element_to_bit(B,p,q)
    E2B --> HE: bit

    HE -> HE: extracted_bits.append(bit)
end

HE --> T: extracted_bits

T -> T: 比较原始比特与提取比特
alt match
    T -> T: 每8位恢复字符
    T -> T: 输出 recovered_msg
else mismatch
    T -> T: 输出验证失败
end

@enduml
```
